{# Small UI macros shared by the shop pages. Usage: {% from "_ui.html" import plate, price %} #} {# Race number plate: a white plate with an ink border and big condensed digits. Only for real identifying numbers (engine class, order number, cart count). `unit` is optional ("cc"). The text stays readable as one string, e.g. "250cc". #} {% macro plate(number, unit=None) -%} {{ number }} {%- if unit %}{{ unit }}{% endif %} {%- endmacro %} {# Price from the database (cents) with small decimals: "12,999" + ".00 EUR". The two spans touch, so the text reads "12,999.00 EUR" for screen readers. #} {% macro price(cents) -%} {{ "{:,}".format(cents // 100) }}.{{ "%02d"|format(cents % 100) }} EUR {%- endmacro %}
